cmake_minimum_required(VERSION 3.20)
project(DcsVrBinaural LANGUAGES CXX)

# Binaural (HRTF) audio for DCS: the interface headers (include/), BinauralCore and its offline harness (task A in
# docs/BINAURAL_AUDIO.md). The audio module in native/dcsqvcull and the pose publisher in native/pupil_shift include
# include/ from here (no copies).
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_compile_options(/W4 /WX /permissive- /EHsc /DUNICODE /D_UNICODE /DWIN32_LEAN_AND_MEAN /DNOMINMAX /Brepro)
add_link_options(/Brepro)

add_library(binaural_headers INTERFACE)
target_include_directories(binaural_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

# BinauralCore: the renderer (include/binaural_core.h). Static, no DCS dependency. Consumers link `binaural_core` and
# get include/ (the interface) on their include path; src/ is private (the harness adds it for its probes).
# kernels_avx2.cpp is the only file built with /arch:AVX2; it runs only when the CPU reports AVX2 and FMA.
add_library(binaural_core STATIC
    src/hrtf_set.cpp
    src/renderer.cpp
    src/kernels_sse.cpp
    src/kernels_avx2.cpp)
target_link_libraries(binaural_core PUBLIC binaural_headers)
target_compile_options(binaural_core PRIVATE /O2 /fp:precise)
set_source_files_properties(src/kernels_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")

enable_testing()

# The contract: every interface header compiles together (with windows.h), layouts hold, pure helpers behave.
add_executable(binaural_contract_test test/contract_test.cpp)
target_link_libraries(binaural_contract_test PRIVATE binaural_headers)
add_test(NAME binaural_contract COMMAND binaural_contract_test)
set_tests_properties(binaural_contract PROPERTIES PASS_REGULAR_EXPRESSION " 0 failure\\(s\\)")

# The BinauralCore harness: tests A1-A13 of docs/BINAURAL_AUDIO.md 8.2 on a synthetic spherical-head set (no download
# needed), the format round trip, and the C#-converter fixture. Listening dumps (WAV/CSV) go to
# BINAURAL_DUMP_DIR when that environment variable is set.
add_executable(binaural_core_test test/core_test.cpp)
target_link_libraries(binaural_core_test PRIVATE binaural_core)
target_include_directories(binaural_core_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR}/test)
target_compile_options(binaural_core_test PRIVATE /O2)
target_compile_definitions(binaural_core_test PRIVATE
    BINAURAL_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/test/data"
    BINAURAL_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
add_test(NAME binaural_core COMMAND binaural_core_test)
set_tests_properties(binaural_core PROPERTIES PASS_REGULAR_EXPRESSION " 0 failure\\(s\\)")

# A14: the bundled generic set (generated by scripts/fetch-hrtf.ps1 + the CLI). Prints "SKIP A14" until it exists.
add_test(NAME binaural_default_data COMMAND binaural_core_test --default-data)
set_tests_properties(binaural_default_data PROPERTIES PASS_REGULAR_EXPRESSION " 0 failure\\(s\\)")
